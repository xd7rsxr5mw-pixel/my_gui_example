cmake_minimum_required(VERSION 3.24)
project(MyGuiExample VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MYGUI_EXAMPLE_BUILD_APP "Build the example application" ON)
option(MYGUI_EXAMPLE_BUILD_TESTS "Build the example's backend and startup tests" OFF)
option(MYGUI_ENGINE_BUILD_TESTS "Build the reusable engine's tests" OFF)

find_package(Qt6 6.9 REQUIRED COMPONENTS Qml Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.9)

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/gui_engine/CMakeLists.txt")
    message(FATAL_ERROR "The engine submodule is missing. Run: git submodule update --init --recursive")
endif()
if(MYGUI_EXAMPLE_BUILD_TESTS OR MYGUI_ENGINE_BUILD_TESTS)
    enable_testing()
endif()
add_subdirectory(gui_engine)
if(MYGUI_EXAMPLE_BUILD_APP OR MYGUI_EXAMPLE_BUILD_TESTS)
    add_subdirectory(backend)
endif()
if(MYGUI_EXAMPLE_BUILD_APP)
    add_subdirectory(gui_src)
endif()
if(MYGUI_EXAMPLE_BUILD_TESTS)
    add_subdirectory(gui_testing)
endif()
